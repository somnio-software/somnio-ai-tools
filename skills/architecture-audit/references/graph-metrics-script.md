# Graph Metrics Script

> The fixed, standard-library-only Python 3.9+ script the architecture
> audit uses for its inventory, profile and dependency-graph metrics. It is
> not a step of its own: steps 1 and 2 cut it out of this file by the two
> marker lines below and run it. Never retype or edit it during an audit.

---

Usage, from the audited repository's root:

- `graph_metrics.py inventory --root . --dir <data dir> --lang dart|ts|py|none --source <root> ... [--exclude <glob>] [--generated <glob>] [--handwritten <glob>] [--bot <glob>]`
  writes `files.tsv` (path, lines, `src` or `gen`) and `inventory.json`
  (totals, the files marked generated by their header, the overrides, and
  the history facts the profile needs, with the bot accounts left out).
- `graph_metrics.py metrics --root . --dir <data dir> --lang dart|ts|py|none [--driver D1 ...] [--profile compact|standard|large]`
  reads `files.tsv` and `module_map.tsv` and writes `edges.tsv` (source
  file, target file, line), `module_edges.tsv` (every resolved internal
  edge under a header row: source file, line, source module and role,
  target file, module and role, and a status of `internal`, `allowed`,
  `deep`, `unchecked`, `unmapped` or the rule it violates) and
  `metrics.json`. A history that git cannot read (no commit yet, a bad
  branch, a git older than 2.31, no git) never stops it: `changeLocality`
  then holds nulls and a `reason`. A graph with no edges and no unresolved
  import is written with a `warning`, not treated as a failure.

The script only reads the repository (including read-only `git` commands)
and writes nothing outside `--dir`; it refuses to write through a symlink,
and refuses a `--dir` that contains `..`. It checks its own text against
`SELF_DIGEST` on every run and exits with code 3 when the two differ. Line
endings are normalised before hashing, so a copy saved with CRLF line endings
still runs.

Maintainers: after changing the script, set `SELF_DIGEST` to the `got`
value of its exit-3 message, then run the suite from the skill directory:
`python3 -m unittest discover -s tests -p "test_*.py"`.

"""Architecture audit measurements for one repository.

  inventory  walk the source roots and write files.tsv and inventory.json
  metrics    read files.tsv and module_map.tsv, extract import edges, and
             write edges.tsv, module_edges.tsv and metrics.json

Outputs, all in --dir:
  files.tsv         path, lines, kind (src or gen); no header row
  inventory.json    totals, the git facts the profile needs, the files only a header
                    marked generated (generatedByHeader) and the path overrides
  edges.tsv         source file, target file, line of every resolved internal
                    import; no header row
  module_edges.tsv  the same edges with their modules, under a header row:
                    source_file, line, source_module, source_role,
                    target_file, target_module, target_role, status. status
                    is internal (both files in one module), allowed, deep
                    (allowed, but past the target's public surface),
                    unchecked (the target declares no surface), the violated
                    rule (R1, R3, R5, R7), or unmapped (a file matches no
                    module row; its module and role are '-')
  metrics.json      the measurements; externalImportsByModule is module ->
                    package -> {files: importing files, example: "file:line"}

Standard library only, Python 3.9+. The audited repository is only read: git
runs read-only commands, and nothing is written outside --dir.

def die(message):
    sys.exit(f'graph_metrics: {message}')


def warn(message):

def norm(path):
    """Repo-relative POSIX path; the repository root itself is ''."""
    path = posixpath.normpath(path.replace(os.sep, '/'))
    return '' if path == '.' else path


def rel_to(root, absolute):
    return norm(os.path.relpath(absolute, root))


def pct(part, whole):
    """A percentage, unrounded: the standard compares it with thresholds, and the report rounds for display."""
    return None if not whole else 100.0 * part / whole


def read_text(root, rel):
    with open(os.path.join(root, rel), encoding='utf-8', errors='replace') as handle:
        return handle.read().lstrip('\ufeff')


def read_tsv(path, columns, comments=False, hint=''):
    """(line number, cells) for every non-blank row of a tab-separated file."""
    if not os.path.isfile(path):
        die(f'{path} not found{hint}')
    rows = []
    with open(path, encoding='utf-8') as handle:
        for number, line in enumerate(handle, 1):
            line = line.rstrip('\r\n')
            if not line.strip() or (comments and line.startswith('#')):
                continue
            cells = line.split('\t')
            if len(cells) != columns:
                die(f'{path}:{number}: expected {columns} tab-separated columns, got {len(cells)}{hint}')
            rows.append((number, cells))
    return rows

def check_output_dir(root, directory, command):
    """Refuse symlinks among the files the command writes, on the way down to --dir inside --root, and any link
    that makes an outside --dir resolve into --root."""
    if os.pardir in directory.split(os.sep):
        die(f'--dir {directory} contains "..", which a symlink before it can redirect; give the path without it')
    root, directory = os.path.abspath(root), os.path.abspath(directory)
    prefix = root.rstrip(os.sep) + os.sep
    if directory.startswith(prefix):
        walked = root
        for part in directory[len(prefix):].split(os.sep):
            walked = os.path.join(walked, part)
            if os.path.islink(walked):
                die(f'{walked} is a symlink; refusing to write through it')
    real_root, real_directory = os.path.realpath(root), os.path.realpath(directory)
    if real_directory == real_root or real_directory.startswith(real_root.rstrip(os.sep) + os.sep):
        # --dir lands in the repository: its lexical path must say so, or a link elsewhere reaches the source tree
        lexical = os.path.relpath(directory, root)
        if lexical.split(os.sep)[0] == '..' or lexical != os.path.relpath(real_directory, real_root):
            die(f'{directory} resolves to {real_directory}, inside --root, through a symlink; refusing to write there')
    for name in OUTPUT_FILES[command]:
        if os.path.islink(os.path.join(directory, name)):
            die(f'{os.path.join(directory, name)} is a symlink; refusing to write through it')


def open_output(path):
    """Text file opened for writing; a symlink at `path` is refused rather than followed."""
    if os.path.islink(path):
        die(f'{path} is a symlink; refusing to write through it')
    flags = os.O_WRONLY | os.O_CREAT | os.O_TRUNC | getattr(os, 'O_NOFOLLOW', 0)
    try:
        descriptor = os.open(path, flags, 0o666)
    except OSError as error:
        die(f'cannot write {path}: {error.strerror or error}')
    return os.fdopen(descriptor, 'w', encoding='utf-8')


def write_json(path, data):
    with open_output(path) as handle:
        json.dump(data, handle, indent=2, sort_keys=True, ensure_ascii=False)
        handle.write('\n')


class LineIndex:
    def __init__(self, text):
        self.starts = [match.end() for match in re.finditer('\n', text)]

    def line(self, offset):
        return bisect.bisect_right(self.starts, offset) + 1


class GitError(Exception):

class GitMissing(GitError):
    pass


def git(root, *args):
    env = dict(os.environ, GIT_OPTIONAL_LOCKS='0')
    # User settings that would add lines to the parsed output are switched off.
    command = ['git', '-c', 'color.ui=never', '-c', 'log.showSignature=false', *args]
    try:
        result = subprocess.run(command, cwd=root, capture_output=True, env=env,
                                encoding='utf-8', errors='replace', stdin=subprocess.DEVNULL)
    except OSError as error:
        raise GitMissing(f'git cannot run: {error}') from error
    if result.returncode:
        raise GitError('; '.join(result.stderr.strip().splitlines()) or f'git {args[0]} failed')
    return result.stdout


def git_unavailable(root, no_git):
    """Why git facts cannot be read here, or None when they can."""
    if no_git:
        return '--no-git'
    try:
        inside = git(root, 'rev-parse', '--is-inside-work-tree').strip()
    except GitMissing as error:
        return str(error)
    except GitError:
        return 'not a git repository'
    if inside != 'true':
        return 'not a git repository'
    try:
        shallow = git(root, 'rev-parse', '--is-shallow-repository').strip()
    except GitError as error:
        return f'git failed: {error}'
    return 'shallow clone' if shallow == 'true' else None


def glob_hits(value, patterns):
    """The patterns (fnmatch, case-insensitive) that match `value`."""
    value = value.lower()
    return [pattern for pattern in patterns if fnmatch.fnmatchcase(value, pattern.lower())]


def is_bot(name, email):
    name, email = name.strip().lower(), email.strip().lower()
    local = email.rpartition('@')[0] if '@' in email else email
    words = name.split()
    return (name.endswith('[bot]') or (bool(words) and words[-1] in ('bot', '(bot)'))
            or local == 'bot' or local.endswith(BOT_LOCAL_SUFFIXES) or GITLAB_BOT_LOCAL.fullmatch(local) is not None
            or BOT_SERVICES.search(name) is not None or BOT_SERVICES.search(local) is not None)


def is_project_dir(root, rel_dir):
    base = os.path.join(root, rel_dir)
    return rel_dir == '' or any(os.path.isfile(os.path.join(base, name)) for name in PROJECT_MANIFESTS)


def skip_dir(root, rel_dir, name, excludes):
    if name in SKIPPED_DIRS:
        return True
    if name in OUTPUT_DIRS and is_project_dir(root, rel_dir):
        return True
    path = posixpath.join(rel_dir, name)
    return any(fnmatch.fnmatchcase(path, pattern) for pattern in excludes)


def find_manifests(root, names):
    """Repo-relative directories holding each named file, in sorted walk order."""
    found = defaultdict(list)
    for dirpath, dirnames, filenames in os.walk(root):
        rel_dir = rel_to(root, dirpath)
        dirnames[:] = sorted(d for d in dirnames if not d.startswith('.') and not skip_dir(root, rel_dir, d, ()))
        for name in names:
            if name in filenames:
                found[name].append(rel_dir)
    return found

def walk_source(root, source, extensions, excludes):
    top = norm(source)
    if top.startswith('../') or top == '..' or os.path.isabs(source):
        die(f'--source {source} must be a path inside --root')
    absolute = os.path.join(root, top)
    if os.path.isfile(absolute):
        yield top
        return
    if not os.path.isdir(absolute):
        die(f'--source {source} does not exist under {root}')
    for dirpath, dirnames, filenames in os.walk(absolute):
        rel_dir = rel_to(root, dirpath)
        # Dot directories (.git, .github, .claude, .dart_tool, ...) hold tooling, not product source.
        dirnames[:] = sorted(d for d in dirnames if not d.startswith('.') and not skip_dir(root, rel_dir, d, excludes))
        for name in sorted(filenames):
            path = posixpath.join(rel_dir, name)
            if (name.endswith(extensions)
                    and not any(fnmatch.fnmatchcase(name, pattern) for pattern in SKIPPED_FILES)
                    and not any(fnmatch.fnmatchcase(path, pattern) for pattern in excludes)):
                yield path


def l10n_outputs(root, directories):
    """(output directory, file stem) of every gen-l10n configuration (l10n.yaml).

    gen-l10n output carries no generated-file header in current Flutter, so it
    is recognised from its configuration instead.
    """
    outputs = []
    for rel_dir in directories:
        settings = {}
        with open(os.path.join(root, rel_dir, 'l10n.yaml'), encoding='utf-8', errors='replace') as handle:
            for line in handle:
                key, separator, value = line.partition(':')
                if separator and not line[:1].isspace() and not line.startswith('#'):
                    settings[key.strip()] = value.split('#')[0].strip().strip('\'"')
        arb_dir = settings.get('arb-dir', 'lib/l10n')
        output_dir = settings.get('output-dir', arb_dir)
        stem = posixpath.splitext(settings.get('output-localization-file', 'app_localizations.dart'))[0]
        outputs.append((norm(posixpath.join(rel_dir, output_dir)), stem))
    return outputs


def inspect_file(path):
    """Physical line count (wc -l, plus an unterminated last line) and the header text."""
    with open(path, 'rb') as handle:
        data = handle.read()
    lines = data.count(b'\n') + (1 if data and not data.endswith(b'\n') else 0)
    head = b'\n'.join(data.split(b'\n', HEADER_LINES)[:HEADER_LINES]).decode('utf-8-sig', 'replace')
    return lines, head


def header_text(head, python):
    """The comment text, decoration stripped and on one line, of the leading comment block and of a Python
    module docstring, up to the first code line."""
    lines, closing = [], None
    for line in head.splitlines()[:HEADER_LINES]:
        if closing is not None:  # inside a block comment or a docstring
            lines.append(COMMENT_LEADER.sub('', COMMENT_TRAILER.sub('', line)) if closing == '*/' else line)
            if closing in line:
                closing = None
            continue
        if not line.strip():
            continue
        comment = HEADER_COMMENT.match(line)
        if comment is not None:
            lines.append(COMMENT_TRAILER.sub('', comment.group(1)))
            opened = line.lstrip()
            if opened.startswith('/*') and '*/' not in opened[2:]:
                closing = '*/'
            continue
        quote = DOCSTRING_QUOTE.match(line) if python else None
        if quote is None:
            break
        rest = line[quote.end():]
        lines.append(rest)
        if quote.group(1) not in rest:
            closing = quote.group(1)
    return re.sub(r'\s+', ' ', ' '.join(lines))


def generated_header(head, python):
    """Whether the header says a generator wrote the file."""
    text = header_text(head, python)
    return GENERATED_MARKER.search(text) is not None or (
        GENERATION_CLAIM.search(text) is not None and DO_NOT_EDIT_CLAIM.search(text) is not None)


def generated_by_path(path, outputs):
    """Whether the file name, a directory or a gen-l10n configuration marks the file generated."""
    directory, name = posixpath.split(path)
    if any(fnmatch.fnmatchcase(name, pattern) for pattern in GENERATED_FILES):
        return True
    if GENERATED_DIRS.intersection(directory.split('/')):
        return True
    return any(directory == out_dir and name.endswith('.dart') and (name == f'{stem}.dart' or name.startswith(f'{stem}_'))
               for out_dir, stem in outputs)


def git_facts(root, no_git, bot_patterns=(), paths=('.',)):
    """Git facts for the profile; botsExcluded stays null with the other facts when history cannot be read.

    Contributors are counted from the commits that touch `paths` (the inventory's sources); the age is repo-wide.
    """
    facts = dict.fromkeys(('isShallow', 'firstCommitDate', 'ageDays', 'peakMonthlyContributors',
                           'contributors90d', 'botsExcluded', 'reason'))
    facts['botPatterns'] = sorted(set(bot_patterns))
    facts['contributorPaths'] = sorted(set(paths))
    reason = git_unavailable(root, no_git)
    facts['isShallow'] = {'shallow clone': True, None: False}.get(reason)
    if reason:
        facts['reason'] = reason
        return facts
    now = dt.datetime.now(dt.timezone.utc)
    try:
        roots = git(root, 'log', '--max-parents=0', '--format=%at', 'HEAD').split()
        months = []
        year, month = now.year, now.month
        for _ in range(12):
            months.append(f'{year:04d}-{month:02d}')
            year, month = (year - 1, 12) if month == 1 else (year, month - 1)
        window = dt.datetime(int(months[-1][:4]), int(months[-1][5:]), 1, tzinfo=dt.timezone.utc)
        log = git(root, 'log', f'--since={window.isoformat()}', '--format=%at%x09%aN%x09%aE', 'HEAD', '--',
                  *(f':(literal){path}' for path in facts['contributorPaths']))
    except GitError as error:
        facts['reason'] = f'git failed: {error}'
        return facts
    first = dt.datetime.fromtimestamp(min(map(int, roots)), dt.timezone.utc)
    facts['firstCommitDate'] = first.isoformat()
    facts['ageDays'] = (now - first).days
    per_month, recent, bots, matched = defaultdict(set), set(), set(), set()
    for row in log.splitlines():
        stamp, _, rest = row.partition('\t')
        name, _, email = rest.partition('\t')
        if not stamp.isdigit():
            continue
        hits = glob_hits(name, facts['botPatterns']) + glob_hits(email, facts['botPatterns'])
        matched.update(hits)
        if hits or is_bot(name, email):
            bots.add(f'{name} <{email}>')
            continue
        when = dt.datetime.fromtimestamp(int(stamp), dt.timezone.utc)
        per_month[when.strftime('%Y-%m')].add(name)
        if now - when <= dt.timedelta(days=90):
            recent.add(name)
    facts['peakMonthlyContributors'] = max(len(per_month[key]) for key in months)
    facts['contributors90d'] = len(recent)
    facts['botsExcluded'] = sorted(bots)
    for pattern in facts['botPatterns']:
        if pattern not in matched:
            warn(f'--bot {pattern} matched no author in the last twelve months')
    return facts


def run_inventory(args):
    root = os.path.abspath(args.root)
    check_output_dir(root, args.dir, 'inventory')
    os.makedirs(args.dir, exist_ok=True)
    if args.lang == 'none':
        if not args.ext:
            die('--lang none needs at least one --ext')
        extensions = tuple(ext if ext.startswith('.') else f'.{ext}' for ext in args.ext)
    else:
        extensions = LANG_EXTENSIONS[args.lang]
    excludes = sorted(set(args.exclude or ()))
    overrides = {'generated': sorted(set(args.generated or ())), 'handwritten': sorted(set(args.handwritten or ()))}
    outputs = l10n_outputs(root, find_manifests(root, ('l10n.yaml',))['l10n.yaml']) if args.lang == 'dart' else []
    paths = sorted({path for source in args.source for path in walk_source(root, source, extensions, excludes)})
    rows, by_header, used = [], [], set()
    for path in paths:
        if '\t' in path or '\n' in path:
            warn(f'skipping {path!r}: a tab or newline in the name cannot be written to files.tsv')
            continue
        lines, head = inspect_file(os.path.join(root, path))
        forced = {kind: [glob for glob in globs if fnmatch.fnmatchcase(path, glob)] for kind, globs in overrides.items()}
        used.update(forced['generated'], forced['handwritten'])
        if forced['generated'] and forced['handwritten']:
            die(f"{path} matches both --generated {forced['generated'][0]} and --handwritten {forced['handwritten'][0]}")
        if forced['generated'] or forced['handwritten']:
            generated = bool(forced['generated'])
        elif generated_by_path(path, outputs):
            generated = True
        else:
            generated = generated_header(head, path.endswith('.py'))
            if generated:
                by_header.append(path)
        rows.append((path, lines, 'gen' if generated else 'src'))
    for option, globs in overrides.items():
        for glob in globs:
            if glob not in used:
                warn(f'--{option} {glob} matches no inventoried file')
    with open_output(os.path.join(args.dir, 'files.tsv')) as handle:
        handle.writelines(f'{path}\t{lines}\t{kind}\n' for path, lines, kind in rows)
    src = [lines for _, lines, kind in rows if kind == 'src']
    gen = [lines for _, lines, kind in rows if kind == 'gen']
    sources = sorted({norm(source) or '.' for source in args.source})
    summary = {
        'files': len(rows), 'srcFiles': len(src), 'genFiles': len(gen),
        'srcLines': sum(src), 'genLines': sum(gen),
        'sources': sources, 'excludes': excludes,
        'generatedByHeader': by_header, 'overrides': overrides,
        'git': git_facts(root, args.no_git, args.bot or (), sources),
    }
    write_json(os.path.join(args.dir, 'inventory.json'), summary)
    print(json.dumps(summary, indent=2, sort_keys=True))


# --- Source scanning ---------------------------------------------------------

_DART_TOKENS = r"""//|/\*|(?<![\w$])r(?:'''|\"\"\"|'|")|'''|\"\"\"|'|\""""
_TS_TOKENS = r"""//|/\*|/|'|"|`"""
SCAN_TOKENS = {
    'dart': (re.compile(_DART_TOKENS), re.compile(_DART_TOKENS + r'|[{}]')),
    'ts': (re.compile(_TS_TOKENS), re.compile(_TS_TOKENS + r'|[{}]')),
}
REGEX_PRECEDERS = frozenset('(,=:[!&|?{};+-*%<>~^')
REGEX_KEYWORDS = frozenset({'return', 'typeof', 'case', 'do', 'else', 'in', 'of', 'new', 'delete',
                            'void', 'throw', 'instanceof', 'yield', 'await'})


@functools.lru_cache(maxsize=None)
def _string_body(quote, raw, interpolates):
    parts = [re.escape(quote)]
    if not raw:
        parts.append(r'\\[\s\S]')
    if interpolates:
        parts.append(r'\$\{')
    if quote in ("'", '"'):
        parts.append(r'\n')  # single-line literal: a newline ends a broken one
    return re.compile('|'.join(parts))


class Scan:
    """A source file with comments blanked out (offsets and newlines kept).

    `ends` maps the offset of each string literal's opening quote to the
    offset just past its closing quote, so a regex match over `code` can
    tell a real directive from text inside a string.
    """

    def __init__(self, text, dialect):
        self.dialect = dialect
        self.ends = {}
        plain, with_braces = SCAN_TOKENS[dialect]
        comments, stack, pos = [], [], 0  # stack: [brace depth, quote, raw, start] per open interpolation
        while True:
            match = (with_braces if stack else plain).search(text, pos)
            if match is None:
                break
            token, start = match.group(), match.start()
            if token == '//':
                end = text.find('\n', start)
                pos = len(text) if end < 0 else end
                comments.append((start, pos))
            elif token == '/*':
                pos = self._block_end(text, start)
                comments.append((start, pos))
            elif token == '/':
                pos = self._regex_end(text, start)
            elif token == '{':
                stack[-1][0] += 1
                pos = match.end()
            elif token == '}':
                if stack[-1][0]:
                    stack[-1][0] -= 1
                    pos = match.end()
                else:
                    _, quote, raw, opened = stack.pop()
                    pos = self._string(text, match.end(), quote, raw, opened, stack)
            else:
                raw = token[0] == 'r'
                quote = token[1:] if raw else token
                opened = start + raw
                pos = self._string(text, opened + len(quote), quote, raw, opened, stack)
        pieces, last = [], 0
        for start, end in comments:
            pieces.append(text[last:start])
            pieces.append(re.sub(r'[^\n]', ' ', text[start:end]))
            last = end
        pieces.append(text[last:])
        self.code = ''.join(pieces)

    def _string(self, text, pos, quote, raw, opened, stack):
        interpolates = not raw and (self.dialect == 'dart' or quote == '`')
        body = _string_body(quote, raw, interpolates)
        while True:
            match = body.search(text, pos)
            if match is None:
                self.ends[opened] = len(text)
                return len(text)
            token = match.group()
            if token == '${':
                stack.append([0, quote, raw, opened])
                return match.end()
            if token == quote:
                self.ends[opened] = match.end()
                return match.end()
            if token == '\n':
                self.ends[opened] = match.start()
                return match.start()
            pos = match.end()

    def _block_end(self, text, start):
        if self.dialect != 'dart':
            end = text.find('*/', start + 2)
            return len(text) if end < 0 else end + 2
        depth, pos = 1, start + 2  # Dart block comments nest
        for match in re.compile(r'/\*|\*/').finditer(text, pos):
            depth += 1 if match.group() == '/*' else -1
            if depth == 0:
                return match.end()
        return len(text)

    @staticmethod
    def _regex_end(text, start):
        """End of a JS regex literal at `start`, or start + 1 when `/` divides."""
        j = start - 1
        while j >= 0 and text[j] in ' \t\r\n':
            j -= 1
        if j >= 0 and text[j] not in REGEX_PRECEDERS:
            k = j
            while k >= 0 and (text[k].isalnum() or text[k] in '_$'):
                k -= 1
            if text[k + 1:j + 1] not in REGEX_KEYWORDS:
                return start + 1
        i, in_class = start + 1, False
        while i < len(text):
            char = text[i]
            if char == '\\':
                i += 2
                continue
            if char == '\n':
                return start + 1
            if char == '[':
                in_class = True
            elif char == ']':
                in_class = False
            elif char == '/' and not in_class:
                return i + 1
            i += 1
        return start + 1

    def literal(self, opened):
        quote = 3 if self.code.startswith(("'''", '"""'), opened) else 1
        return self.code[opened + quote:self.ends[opened] - quote]


def previous_char(code, offset):
    j = offset - 1
    while j >= 0 and code[j].isspace():
        j -= 1
    return code[j] if j >= 0 else ''

@dataclass
class Site:
    line: int
    public: bool
    type_only: bool


@dataclass
class Extraction:
    edges: dict = field(default_factory=dict)  # (source, target) -> Site
    unresolved: set = field(default_factory=set)  # (source, line, specifier)
    externals: dict = field(default_factory=lambda: defaultdict(dict))  # source -> {package: first line}
    platform: dict = field(default_factory=lambda: defaultdict(set))
    parse_errors: list = field(default_factory=list)

    def external(self, source, package, line):
        known = self.externals[source].get(package)
        self.externals[source][package] = line if known is None else min(known, line)

    def link(self, source, target, line, public=False, type_only=False):
        if source == target:
            return
        site = self.edges.get((source, target))
        if site is None:
            self.edges[(source, target)] = Site(line, public, type_only)
        else:
            site.line = min(site.line, line)
            site.public = site.public or public
            site.type_only = site.type_only and type_only

def dart_packages(root):
    """Package name -> its lib/ directory, from every pubspec.yaml under root."""
    packages = {}
    for rel_dir in find_manifests(root, ('pubspec.yaml',))['pubspec.yaml']:
        match = re.search(r"""^name:\s*['"]?([A-Za-z_]\w*)""", read_text(root, posixpath.join(rel_dir, 'pubspec.yaml')),
                          re.MULTILINE)
        if match and match.group(1) not in packages:
            packages[match.group(1)] = posixpath.join(rel_dir, 'lib')
    return packages


def dart_uris(scan, opened):
    """The directive's URI and every conditional URI, up to its `;`."""
    code, ends = scan.code, scan.ends
    uris, pos = [scan.literal(opened)], ends[opened]
    while True:
        match = STATEMENT_STOP.search(code, pos)
        if match is None or match.group() == ';':
            return uris
        quote = match.start()
        if quote not in ends:
            pos = quote + 1
            continue
        before = quote - 1 if code[quote - 1:quote] == 'r' else quote
        if previous_char(code, before) == ')':  # `if (dart.library.io) 'io.dart'`
            uris.append(scan.literal(quote))
        pos = ends[quote]


def link_or_report(out, root, files, source, line, specifier, target):
    if target in files:
        out.link(source, target, line)
    elif not os.path.isfile(os.path.join(root, target)):
        out.unresolved.add((source, line, specifier))


def extract_dart(root, files):
    out, packages = Extraction(), dart_packages(root)
    for source in sorted(files):
        scan = Scan(read_text(root, source), 'dart')
        lines = LineIndex(scan.code)
        for match in DART_DIRECTIVE.finditer(scan.code):
            if match.end() not in scan.ends:
                continue
            line = lines.line(match.start())
            for uri in dart_uris(scan, match.end()):
                if uri.startswith('dart:'):
                    if uri in PLATFORM_LIBRARIES:
                        out.platform[source].add(uri)
                elif uri.startswith('package:'):
                    name, _, rest = uri[len('package:'):].partition('/')
                    if name in packages:
                        link_or_report(out, root, files, source, line, uri, norm(posixpath.join(packages[name], rest)))
                    else:
                        out.external(source, name, line)
                elif not URI_SCHEME.match(uri):
                    target = norm(posixpath.join(posixpath.dirname(source), uri))
                    link_or_report(out, root, files, source, line, uri, target)
    return out

def parse_jsonc(text):
    text = JSONC_COMMENT.sub(lambda match: match.group(1) or '', text)
    return json.loads(JSONC_TRAILING_COMMA.sub(lambda match: match.group(1) or '', text))


def package_name(specifier):
    parts = specifier.split('/')
    return '/'.join(parts[:2]) if specifier.startswith('@') else parts[0]


@dataclass
class TsConfig:
    base_url: str | None
    paths: list  # (pattern, [repo-relative targets with the `*` kept])

    def match(self, specifier):
        """Targets of the alias that applies to `specifier` (exact, else longest prefix)."""
        best = None
        for pattern, targets in self.paths:
            if '*' not in pattern:
                if pattern == specifier:
                    return targets
                continue
            prefix, suffix = pattern.split('*', 1)
            if (specifier.startswith(prefix) and specifier.endswith(suffix)
                    and len(specifier) >= len(prefix) + len(suffix)
                    and (best is None or len(prefix) > len(best[0]))):
                best = (prefix, suffix, targets)
        if best is None:
            return None
        prefix, suffix, targets = best
        star = specifier[len(prefix):len(specifier) - len(suffix)]
        return [norm(target.replace('*', star, 1)) for target in targets]


class TsResolver:
    UNRESOLVED, SKIP = ('unresolved',), ('skip',)

    def __init__(self, root, files):
        self.root, self.files = root, files
        self.packages, self.manifests = {}, {}
        for rel_dir in find_manifests(root, ('package.json',))['package.json']:
            try:
                manifest = json.loads(read_text(root, posixpath.join(rel_dir, 'package.json')))
            except json.JSONDecodeError as error:
                warn(f'{posixpath.join(rel_dir, "package.json")}: {error}; package ignored')
                continue
            if isinstance(manifest, dict):
                self.manifests[rel_dir] = manifest
            name = manifest.get('name') if isinstance(manifest, dict) else None
            if isinstance(name, str) and name and name not in self.packages:
                self.packages[name] = (rel_dir, manifest)
        self._raw_configs, self._nearest = {}, {}

    def candidates(self, base):
        stem, extension = posixpath.splitext(base)
        if base:
            yield base
        for alternative in ESM_SOURCES.get(extension, ()):
            yield stem + alternative
        for extension in TS_EXTENSIONS:
            if base:
                yield base + extension
            yield posixpath.join(base, 'index' + extension)

    def find(self, base):
        return next((path for path in self.candidates(base) if path in self.files), None)

    def on_disk(self, base):
        extra = (base + '.d.ts', posixpath.join(base, 'index.d.ts'))
        return any(os.path.isfile(os.path.join(self.root, path)) for path in (*self.candidates(base), *extra))

    def local(self, base):
        target = self.find(base)
        if target:
            return ('edge', target, False)
        return self.SKIP if self.on_disk(base) else self.UNRESOLVED

    def resolve(self, specifier, source):
        """('edge', target, public) | ('external', name) | UNRESOLVED | SKIP."""
        specifier = specifier.split('?', 1)[0]
        if not specifier or specifier.startswith('/'):
            return self.SKIP
        if specifier in ('.', '..') or specifier.startswith(('./', '../')):
            return self.local(norm(posixpath.join(posixpath.dirname(source), specifier)))
        config = self.config_for(posixpath.dirname(source))
        if config is not None:
            targets = config.match(specifier)
            if targets is not None:
                for base in targets:
                    target = self.find(base)
                    if target:
                        return ('edge', target, self.declared_public(specifier))
                return self.SKIP if any(self.on_disk(base) for base in targets) else self.UNRESOLVED
            if config.base_url is not None:
                target = self.find(norm(posixpath.join(config.base_url, specifier)))
                if target:
                    return ('edge', target, self.declared_public(specifier))
        if specifier.startswith('#'):  # a package.json `imports` key, once no tsconfig alias maps it
            return self.package_import(specifier, source)
        name = package_name(specifier)
        if name in self.packages:
            directory, manifest = self.packages[name]
            if specifier == name:
                target = self.entry(directory, manifest)
                return ('edge', target, True) if target else self.UNRESOLVED
            sub = specifier[len(name) + 1:]
            # A subpath the package declares in `exports` is public, whatever the module map says.
            exported = self._subpath_targets(manifest.get('exports'), f'./{sub}')
            for base in exported:
                target = self.find(norm(posixpath.join(directory, base)))
                if target:
                    return ('edge', target, True)
            # Declared targets in build output (dist/) do not resolve to source, so the plain path decides
            # which file it is; the declaration still makes it public.
            for base in (posixpath.join(directory, sub), posixpath.join(directory, 'src', sub)):
                target = self.find(norm(base))
                if target:
                    return ('edge', target, bool(exported))
            return self.SKIP if self.on_disk(norm(posixpath.join(directory, sub))) else self.UNRESOLVED
        if specifier.startswith('node:'):
            return ('platform', specifier)
        if name in NODE_PLATFORM_BUILTINS:  # the builtin or a subpath of it (fs/promises)
            return ('platform', f'node:{specifier}')
        if ':' in specifier or name in NODE_BUILTINS:
            return self.SKIP
        return ('external', name)

    def package_import(self, specifier, source):
        """A `#` specifier through the `imports` of the nearest package.json (the package scope); never public."""
        directory = posixpath.dirname(source)
        while directory not in self.manifests:
            if not directory:
                return self.UNRESOLVED
            directory = posixpath.dirname(directory)
        for target in self._subpath_targets(self.manifests[directory].get('imports'), specifier):
            if target.startswith(('./', '../')):
                result = self.local(norm(posixpath.join(directory, target)))
            elif target.startswith(('#', '/')):
                continue
            else:
                result = self.resolve(target, source)  # a package the mapping hands the import to
            if result is not self.UNRESOLVED:
                return ('edge', result[1], False) if result[0] == 'edge' else result
        return self.UNRESOLVED

    def declared_public(self, specifier):
        """Whether the `exports` of the workspace package a package-form specifier names declare it.

        False when the package has no `exports` or the specifier names no workspace package (an alias such as
        `@/components/x`): the module map's surface decides then.
        """
        name = package_name(specifier)
        exports = self.packages[name][1].get('exports') if name in self.packages else None
        if exports is None:
            return False
        if specifier == name:
            return any(True for _ in self._export_targets(exports))
        return any(True for _ in self._subpath_targets(exports, f'./{specifier[len(name) + 1:]}'))

    def entry(self, directory, manifest):
        """The source file a bare import of a workspace package lands on."""
        candidates = list(self._export_targets(manifest.get('exports')))
        candidates += [manifest[key] for key in ('types', 'module', 'main') if isinstance(manifest.get(key), str)]
        candidates += ['index', 'src/index']
        for candidate in candidates:
            target = self.find(norm(posixpath.join(directory, candidate)))
            if target:
                return target
        return None

    def _export_targets(self, exports):
        if isinstance(exports, dict) and any(key.startswith('.') for key in exports):
            exports = exports.get('.')
        return self._conditions(exports)

    def _subpath_targets(self, exports, key):
        """Targets a declared `exports` (or `imports`) key gives `key`: exact key first, else the longest `*` pattern."""
        if not isinstance(exports, dict) or not any(name.startswith(('.', '#')) for name in exports):
            return []
        if key in exports:
            return list(self._conditions(exports[key]))
        best = None
        for pattern, value in exports.items():
            if pattern.count('*') != 1:
                continue
            prefix, suffix = pattern.split('*')
            if (key.startswith(prefix) and key.endswith(suffix) and len(key) >= len(prefix) + len(suffix)
                    and (best is None or len(prefix) > len(best[0]))):
                best = (prefix, suffix, value)
        if best is None:
            return []
        prefix, suffix, value = best
        star = key[len(prefix):len(key) - len(suffix)]
        return [target.replace('*', star) for target in self._conditions(value)]

    def _conditions(self, value):
        """File targets of one exports value: a path, a fallback list or a conditions object."""
        if isinstance(value, str):
            yield value
        elif isinstance(value, list):
            for item in value:
                yield from self._conditions(item)
        elif isinstance(value, dict):
            for condition in ('types', 'import', 'default', 'require'):
                if condition in value:
                    yield from self._conditions(value[condition])

    def config_for(self, directory):
        """The nearest tsconfig.json or jsconfig.json at or above `directory`."""
        if directory in self._nearest:
            return self._nearest[directory]
        config = None
        for name in ('tsconfig.json', 'jsconfig.json'):
            path = posixpath.join(directory, name)
            if os.path.isfile(os.path.join(self.root, path)):
                raw = self._load(path, frozenset())
                paths_base = raw['base_url'] if raw['base_url'] is not None else raw['paths_dir']
                aliases = [(pattern, [norm(posixpath.join(paths_base, t)) for t in targets if isinstance(t, str)])
                           for pattern, targets in (raw['paths'] or {}).items() if isinstance(targets, list)]
                config = TsConfig(raw['base_url'], aliases)
                break
        else:
            if directory:
                config = self.config_for(posixpath.dirname(directory))
        self._nearest[directory] = config
        return config

    def _load(self, path, seen):
        """compilerOptions after `extends`, with the directories their paths are relative to.

        As in tsc: baseUrl is relative to the config that sets it; paths are
        relative to the effective baseUrl, else to the config that sets them.
        """
        if path in self._raw_configs:
            return self._raw_configs[path]
        raw = {'base_url': None, 'paths': None, 'paths_dir': None}
        try:
            data = parse_jsonc(read_text(self.root, path))
        except json.JSONDecodeError as error:
            warn(f'{path}: {error}; its path aliases are ignored')
            data = {}
        directory = posixpath.dirname(path)
        parents = data.get('extends', [])
        for parent in [parents] if isinstance(parents, str) else parents:
            if not isinstance(parent, str) or not parent.startswith(('./', '../')):
                continue  # a package config from node_modules carries no workspace aliases
            parent_path = norm(posixpath.join(directory, parent))
            if not parent_path.endswith('.json') and not os.path.isfile(os.path.join(self.root, parent_path)):
                parent_path += '.json'
            if parent_path not in seen and os.path.isfile(os.path.join(self.root, parent_path)):
                inherited = self._load(parent_path, seen | {path})
                raw.update({key: value for key, value in inherited.items() if value is not None})
        options = data.get('compilerOptions') or {}
        if isinstance(options.get('baseUrl'), str):
            raw['base_url'] = norm(posixpath.join(directory, options['baseUrl']))
        if isinstance(options.get('paths'), dict):
            raw['paths'], raw['paths_dir'] = options['paths'], directory
        self._raw_configs[path] = raw
        return raw


def extract_ts(root, files):
    out, resolver = Extraction(), TsResolver(root, files)
    for source in sorted(files):
        scan = Scan(read_text(root, source), 'ts')
        lines = LineIndex(scan.code)
        for match in TS_SPECIFIER.finditer(scan.code):
            if match.end() not in scan.ends:
                continue
            specifier, line = scan.literal(match.end()), lines.line(match.start())
            if scan.code[match.end()] == '`' and '${' in specifier:
                continue  # a computed specifier names no file
            result = resolver.resolve(specifier, source)
            if result[0] == 'edge':
                out.link(source, result[1], line, public=result[2])
            elif result[0] == 'platform':
                out.platform[source].add(result[1])
            elif result[0] == 'external':
                out.external(source, result[1], line)
            elif result is TsResolver.UNRESOLVED:
                out.unresolved.add((source, line, specifier))
    return out


def is_stdlib(name):
    names = getattr(sys, 'stdlib_module_names', None)  # Python 3.10+
    return name in names if names is not None else _stdlib_by_location(name)


@functools.lru_cache(maxsize=None)
def _stdlib_by_location(name):
    if name in sys.builtin_module_names:
        return True
    try:
        spec = importlib.util.find_spec(name)
    except (ImportError, ValueError):
        return False
    if spec is None or not spec.origin or spec.origin in ('built-in', 'frozen'):
        return spec is not None
    stdlib = os.path.realpath(sysconfig.get_paths()['stdlib'])
    origin = os.path.realpath(spec.origin)
    return origin.startswith(stdlib + os.sep) and 'site-packages' not in origin


def python_imports(tree):
    """(import node, type_only) for every import, including those under TYPE_CHECKING."""
    found, stack = [], [(tree, False)]
    while stack:
        node, type_only = stack.pop()
        if isinstance(node, (ast.Import, ast.ImportFrom)):
            found.append((node, type_only))
        elif isinstance(node, ast.If) and _is_type_checking(node.test):
            stack.extend((statement, True) for statement in node.body)
            stack.extend((statement, type_only) for statement in node.orelse)
        else:
            stack.extend((child, type_only) for child in ast.iter_child_nodes(node))
    return sorted(found, key=lambda item: (item[0].lineno, item[0].col_offset))


def _is_type_checking(test):
    return ((isinstance(test, ast.Name) and test.id == 'TYPE_CHECKING')
            or (isinstance(test, ast.Attribute) and test.attr == 'TYPE_CHECKING'))


class PyResolver:
    def __init__(self, root, files):
        self.files = files
        manifests = find_manifests(root, ('pyproject.toml', 'setup.cfg', 'setup.py'))
        projects = {''} | {d for dirs in manifests.values() for d in dirs}
        self.roots = sorted(projects | {posixpath.join(d, 'src') for d in projects
                                        if os.path.isdir(os.path.join(root, d, 'src'))})
        self.dirs = {''}
        for path in files:
            parts = path.split('/')[:-1]
            self.dirs.update('/'.join(parts[:i]) for i in range(1, len(parts) + 1))
        self.internal = set()
        for base in self.roots:
            for path in files:
                if not base or path.startswith(base + '/'):
                    first = path[len(base) + 1 if base else 0:].split('/')[0]
                    self.internal.add(first[:-3] if first.endswith('.py') else first)

    @functools.lru_cache(maxsize=None)
    def roots_for(self, directory):
        """Resolution roots: those enclosing the importer, deepest first, then the rest."""
        enclosing = [r for r in self.roots if not r or directory == r or directory.startswith(r + '/')]
        enclosing.sort(key=len, reverse=True)
        return enclosing + [r for r in self.roots if r not in enclosing]

    def module_file(self, stem):
        for candidate in (posixpath.join(stem, '__init__.py'), stem + '.py' if stem else None):
            if candidate in self.files:
                return candidate
        return None

    def handle(self, node, source, type_only, out):
        line = node.lineno
        if isinstance(node, ast.Import):
            for alias in node.names:
                self._absolute(alias.name, source, line, type_only, out)
            return
        names = [alias.name for alias in node.names]
        module = node.module or ''
        if node.level:
            base = posixpath.dirname(source)
            for _ in range(node.level - 1):
                base = posixpath.dirname(base) if base else None
                if base is None:
                    break
            if base is None or not self._from(source, [base], module, names, line, type_only, out):
                out.unresolved.add((source, line, '.' * node.level + module))
        elif not self._from(source, self.roots_for(posixpath.dirname(source)), module, names, line, type_only, out):
            self._outside(module, source, line, out)

    def _absolute(self, dotted, source, line, type_only, out):
        for base in self.roots_for(posixpath.dirname(source)):
            stem = posixpath.join(base, dotted.replace('.', '/'))
            target = self.module_file(stem)
            if target:
                out.link(source, target, line, type_only=type_only)
                return
            if stem in self.dirs:
                return  # namespace package: nothing to point at
        self._outside(dotted, source, line, out)

    def _from(self, source, bases, module, names, line, type_only, out):
        """`from module import names`: an edge per submodule, one to the module for the rest."""
        for base in bases:
            stem = posixpath.join(base, module.replace('.', '/')) if module else base
            module_file = self.module_file(stem)
            if module_file is None and stem not in self.dirs:
                continue
            needs_module = False
            for name in names:
                submodule = None if name == '*' else self.module_file(posixpath.join(stem, name))
                if submodule:
                    out.link(source, submodule, line, type_only=type_only)
                else:
                    needs_module = True
            if needs_module and module_file:
                out.link(source, module_file, line, type_only=type_only)
            return True
        return False

    def _outside(self, dotted, source, line, out):
        top = dotted.split('.')[0]
        if top in self.internal:
            out.unresolved.add((source, line, dotted))
        elif top in PYTHON_PLATFORM_MODULES:
            out.platform[source].add(top)
        elif top != '__future__' and not is_stdlib(top):
            out.external(source, top, line)


def extract_py(root, files):
    out, resolver = Extraction(), PyResolver(root, files)
    for source in sorted(files):
        with open(os.path.join(root, source), 'rb') as handle:
            data = handle.read()
        try:
            with warnings.catch_warnings():
                warnings.simplefilter('ignore')  # invalid escape sequences and the like
                tree = ast.parse(data, filename=source)
        except (SyntaxError, ValueError, RecursionError) as error:
            out.parse_errors.append(f'{source}: {type(error).__name__}: {getattr(error, "msg", error)}')
            continue
        for node, type_only in python_imports(tree):
            resolver.handle(node, source, type_only, out)
    return out

@dataclass
class Module:
    name: str
    role: str
    physical: bool
    surface: str  # '-', 'nonsrc' or 'list'
    public_files: set
    paths: list


class ModuleMap:
    def __init__(self, modules, by_path):
        self.modules = modules
        self.by_path = {path: name for path, name in by_path.items() if path}
        self.catch_all = by_path.get('')
        self._owners = {}

    def owner(self, path):
        """(module, matching row path) by longest segment prefix; '' matches last."""
        if path not in self._owners:
            found, probe = None, path
            while probe:
                if probe in self.by_path:
                    found = (self.modules[self.by_path[probe]], probe)
                    break
                probe = probe.rpartition('/')[0]
            if found is None and self.catch_all is not None:
                found = (self.modules[self.catch_all], '')
            self._owners[path] = found
        return self._owners[path]

    @staticmethod
    def is_public(module, row_path, target):
        if module.surface == 'list':
            return target in module.public_files
        inner = target[len(row_path) + 1:] if row_path else target
        return 'src' not in inner.split('/')[:-1]


def load_module_map(directory, files):
    errors, modules, by_path = [], {}, {}
    for number, cells in read_tsv(os.path.join(directory, 'module_map.tsv'), 5, comments=True):
        name, role, physical, path, public = (cell.strip() for cell in cells)
        where = f'module_map.tsv:{number}'
        if not name:
            errors.append(f'{where}: empty module name')
        if role not in ROLES:
            errors.append(f'{where}: module {name} has unknown role {role!r}; use one of {", ".join(ROLES)}')
        if physical not in ('yes', 'no'):
            errors.append(f"{where}: physical must be exactly 'yes' or 'no', got {physical!r}")
        row_path = norm(path) if path not in ('', '.') else ''
        if posixpath.isabs(row_path) or row_path == '..' or row_path.startswith('../'):
            errors.append(f'{where}: path {path!r} is not repo-relative')
        if row_path in by_path:
            errors.append(f'{where}: path {path or "."!r} repeats the row of module {by_path[row_path]}')
        by_path.setdefault(row_path, name)
        if public in ('-', 'nonsrc'):
            surface, listed = public, set()
        elif public:
            surface, listed = 'list', {norm(item.strip()) for item in public.split(';') if item.strip()}
        else:
            errors.append(f"{where}: public is empty; use '-' for an undeclared surface")
            surface, listed = '-', set()
        known = modules.get(name)
        if known is None:
            modules[name] = Module(name, role, physical == 'yes', surface, listed, [row_path])
            continue
        if (known.role, known.physical) != (role, physical == 'yes'):
            errors.append(f'{where}: module {name} repeats with a different role or physical value')
        if known.surface != surface:
            errors.append(f'{where}: module {name} mixes public surfaces {known.surface!r} and {surface!r}')
        known.public_files |= listed
        known.paths.append(row_path)
    roots = sorted(name for name, module in modules.items() if module.role == 'root')
    if len(roots) > 1:
        errors.append(f"modules {', '.join(roots)} all have role root; the composition root is one module, "
                      "so give every root row the name 'root'")
    if errors:
        die('module_map.tsv is invalid:\n  ' + '\n  '.join(errors))
    modmap = ModuleMap(modules, by_path)
    under_src = set()
    for path in files:
        owner = modmap.owner(path)
        if owner and owner[0].surface == 'nonsrc' and not ModuleMap.is_public(owner[0], owner[1], path):
            under_src.add(owner[0].name)
    for module in modules.values():
        if module.surface == 'nonsrc' and not module.physical and module.name not in under_src:
            errors.append(f"module {module.name}: public 'nonsrc' on a logical module (physical=no) needs files "
                          'under a src/ directory, and none of its files is under one, so every import of it would '
                          "count as public; list its public files instead, or use '-'")
        for path in sorted(module.public_files):
            owner = modmap.owner(path)
            if posixpath.isabs(path) or path.startswith('../'):
                errors.append(f'module {module.name}: public file {path} is not repo-relative')
            elif path not in files:
                errors.append(f'module {module.name}: public file {path} is not in files.tsv '
                              '(public lists hold repo-relative paths)')
            elif owner is None or owner[0] is not module:
                errors.append(f'module {module.name}: public file {path} belongs to another module')
    if errors:
        die('module_map.tsv is invalid:\n  ' + '\n  '.join(errors))
    used = {owner[1] for owner in map(modmap.owner, files) if owner}
    for row_path, name in sorted(by_path.items()):
        if row_path not in used:
            warn(f'module {name}: row path {row_path or "."!r} owns no file in files.tsv')
    return modmap


def load_files(directory):
    files = {}
    hint = ' (files.tsv is path, lines, kind: run the inventory subcommand)'
    for number, (path, lines, kind) in read_tsv(os.path.join(directory, 'files.tsv'), 3, hint=hint):
        where = f'files.tsv:{number}'
        if not lines.isdigit():
            die(f'{where}: line count {lines!r} is not a number')
        if kind not in ('src', 'gen'):
            die(f"{where}: kind must be 'src' or 'gen', got {kind!r}")
        path = norm(path)
        if path in files:
            die(f'{where}: duplicate path {path}')
        files[path] = (int(lines), kind)
    return files


# --- Metrics -----------------------------------------------------------------


def strongly_connected(nodes, graph):
    """Strongly connected components of more than one node, largest first."""
    index, low, stack, on_stack, result = {}, {}, [], set(), []
    for start in sorted(nodes):
        if start in index:
            continue
        index[start] = low[start] = len(index)
        stack.append(start)
        on_stack.add(start)
        work = [(start, iter(sorted(n for n in graph.get(start, ()) if n in nodes)))]
        while work:
            node, children = work[-1]
            advanced = False
            for child in children:
                if child not in index:
                    index[child] = low[child] = len(index)
                    stack.append(child)
                    on_stack.add(child)
                    work.append((child, iter(sorted(n for n in graph.get(child, ()) if n in nodes))))
                    advanced = True
                    break
                if child in on_stack:
                    low[node] = min(low[node], index[child])
            if advanced:
                continue
            work.pop()
            if work:
                low[work[-1][0]] = min(low[work[-1][0]], low[node])
            if low[node] == index[node]:
                component = []
                while True:
                    member = stack.pop()
                    on_stack.discard(member)
                    component.append(member)
                    if member == node:
                        break
                if len(component) > 1:
                    result.append(sorted(component))
    return sorted(result, key=lambda component: (-len(component), component))


def violation_kind(from_role, to_role):
    if from_role == 'feature' and to_role == 'feature':
        return 'feature->feature'
    return f'{from_role}->{to_role}' if to_role in FORBIDDEN.get(from_role, ()) else None


def rule_of(kind):
    if kind.endswith('->root'):
        return 'R5'
    return {'feature->feature': 'R1', 'contract->feature': 'R3'}.get(kind, 'R7')


@dataclass
class CrossEdge:
    source: str
    target: str
    line: int
    origin: Module
    destination: Module
    kind: str | None  # violation kind, None when the edge is allowed
    surface: str  # 'unchecked', 'public' or 'deep'

    @property
    def site(self):
        return f'{self.source}:{self.line} -> {self.target}'

    @property
    def status(self):
        """The edge's module_edges.tsv status: its violated rule, else unchecked, deep or allowed."""
        if self.kind:
            return rule_of(self.kind)
        return self.surface if self.surface in ('unchecked', 'deep') else 'allowed'


def cross_module_edges(extraction, modmap):
    edges = []
    for (source, target), site in sorted(extraction.edges.items(), key=lambda kv: (kv[0][0], kv[1].line, kv[0][1])):
        origin, destination = modmap.owner(source), modmap.owner(target)
        if origin is None or destination is None or origin[0] is destination[0]:
            continue
        module, row_path = destination
        if module.surface == '-':
            surface = 'unchecked'
        elif site.public or ModuleMap.is_public(module, row_path, target):
            surface = 'public'
        else:
            surface = 'deep'
        edges.append(CrossEdge(source, target, site.line, origin[0], module,
                               violation_kind(origin[0].role, module.role), surface))
    return edges


def ranked(counter, limit=10):
    return [[name, count] for name, count in sorted(counter.items(), key=lambda kv: (-kv[1], kv[0]))[:limit] if count]


def coupling_metrics(edges, modmap):
    features = sorted(m.name for m in modmap.modules.values() if m.role == 'feature')
    pairs = {(e.origin.name, e.destination.name): e.kind for e in edges}
    graph, allowed = defaultdict(set), defaultdict(set)
    for (a, b), kind in pairs.items():
        graph[a].add(b)
        if kind is None:
            allowed[a].add(b)
    feature_pairs = sorted((a, b) for (a, b), kind in pairs.items() if kind == 'feature->feature')
    feature_graph = defaultdict(set)
    for a, b in feature_pairs:
        feature_graph[a].add(b)
    names = set(modmap.modules)
    cycles = strongly_connected(names, allowed)
    permitted = [e for e in edges if e.kind is None]
    checked = [e for e in permitted if e.surface != 'unchecked']
    deep = [e for e in checked if e.surface == 'deep']
    coupling = {
        'featureEdges': len(feature_pairs),
        'featureEdgeSites': sum(e.kind == 'feature->feature' for e in edges),
        'featureEdgesPerFeature': len(feature_pairs) / len(features) if features else None,
        'mutualFeaturePairs': [f'{a}<->{b}' for a, b in feature_pairs if a < b and (b, a) in pairs],
        'featureCycles': strongly_connected(set(features), feature_graph),
        'cycles': cycles,
        'largestCycle': len(cycles[0]) if cycles else 0,
        'cyclesAllEdges': strongly_connected(names, graph),
        'checkedEdges': len(checked),
        'deepEdges': len(deep),
        'deepEdgeShare': pct(len(deep), len(checked)),
        'deepEdgesAllEdges': sum(e.surface == 'deep' for e in edges),
        'uncheckedEdgeShare': pct(sum(e.surface == 'unchecked' for e in permitted), len(permitted)),
    }
    violations = defaultdict(list)
    for edge in edges:
        if edge.kind:
            violations[edge.kind].append(edge)
    report = {kind: {'rule': rule_of(kind), 'count': len(found),
                     'modulePairs': sorted({f'{e.origin.name}->{e.destination.name}' for e in found}),
                     'examples': [e.site for e in found[:10]]}
              for kind, found in violations.items()}
    fan_out = Counter({a: len(targets) for a, targets in graph.items()})
    fan_in = Counter(b for _, b in pairs)
    return coupling, report, {
        'featureEdgeList': [f'{a}->{b}' for a, b in feature_pairs],
        'deepEdgeExamples': [e.site for e in deep[:20]],
        'fanOut': ranked(fan_out),
        'fanIn': ranked(fan_in),
    }


def module_edge_rows(extraction, modmap, edges):
    """module_edges.tsv rows: every resolved internal edge with both files' modules and a status."""
    cross = {(edge.source, edge.target): edge.status for edge in edges}
    rows = []
    for (source, target), site in extraction.edges.items():
        origin, destination = modmap.owner(source), modmap.owner(target)
        if origin is None or destination is None:
            status = 'unmapped'
        elif origin[0] is destination[0]:
            status = 'internal'
        else:
            status = cross[(source, target)]
        ends = [(owner[0].name, owner[0].role) if owner else ('-', '-') for owner in (origin, destination)]
        rows.append((source, site.line, *ends[0], target, *ends[1], status))
    return sorted(rows, key=lambda row: (row[0], row[1], row[4]))


def module_table(files, modmap, edges):
    """Every module's role, physical flag, file and line counts, and distinct-module fan-in and fan-out."""
    table = {name: {'role': module.role, 'physical': module.physical, 'files': 0, 'genFiles': 0, 'lines': 0,
                    'fanIn': 0, 'fanOut': 0}
             for name, module in modmap.modules.items()}
    for path, (lines, kind) in files.items():
        owner = modmap.owner(path)
        if owner is None:
            continue
        row = table[owner[0].name]
        if kind == 'src':
            row['files'] += 1
            row['lines'] += lines
        else:
            row['genFiles'] += 1
    for a, b in {(edge.origin.name, edge.destination.name) for edge in edges}:
        table[a]['fanOut'] += 1
        table[b]['fanIn'] += 1
    return table


def size_metrics(files, modmap):
    lines_by_module = {name: 0 for name in modmap.modules}
    total = 0
    for path, (lines, kind) in files.items():
        owner = modmap.owner(path)
        if kind == 'src':
            total += lines
            if owner:
                lines_by_module[owner[0].name] += lines
    modules = modmap.modules.values()
    features = sorted((m for m in modules if m.role == 'feature'), key=lambda m: (-lines_by_module[m.name], m.name))
    feature_lines = sum(lines_by_module[m.name] for m in features)
    largest_feature = None
    if features:
        top = lines_by_module[features[0].name]
        largest_feature = {
            'name': features[0].name, 'lines': top, 'share': pct(top, feature_lines),
            'ratioToMean': top * len(features) / feature_lines if len(features) >= 2 and feature_lines else None,
        }
    shared = sorted((m for m in modules if m.role in SHARED_ROLES), key=lambda m: (-lines_by_module[m.name], m.name))
    largest_shared = None
    if shared:
        top = lines_by_module[shared[0].name]
        largest_shared = {'name': shared[0].name, 'role': shared[0].role, 'lines': top, 'share': pct(top, total)}
    src = sorted(((lines, path) for path, (lines, kind) in files.items() if kind == 'src'), key=lambda x: (-x[0], x[1]))
    over_1000 = [path for lines, path in src if lines > 1000]
    return {
        'linesByModule': lines_by_module,
        'rootShare': pct(sum(lines_by_module[m.name] for m in modules if m.role == 'root'), total),
        'largestFeature': largest_feature,
        'largestSharedModule': largest_shared,
        'physicalFeatureShare': pct(sum(m.physical for m in features), len(features)),
        'filesOver1000': over_1000,
        'filesOver2000': [path for lines, path in src if lines > 2000],
        'filesOver1000Share': pct(len(over_1000), len(src)),
    }


def platform_imports(per_file, modmap):
    """Module -> {platform library: files importing it}."""
    counts = defaultdict(Counter)
    for path, libraries in per_file.items():
        owner = modmap.owner(path)
        if owner:
            counts[owner[0].name].update(libraries)
    return {name: dict(libraries) for name, libraries in counts.items()}


def external_imports(per_file, modmap):
    """Module -> external package -> {files: importing files, example: the first "file:line" in sorted order}."""
    found = defaultdict(dict)
    for path in sorted(per_file):
        owner = modmap.owner(path)
        if owner is None:
            continue
        for package, line in sorted(per_file[path].items()):
            entry = found[owner[0].name].setdefault(package, {'files': 0, 'example': f'{path}:{line}'})
            entry['files'] += 1
    return dict(found)


def unquote_git_path(text):
    """Undo git's C-style quoting of unusual paths ("caf\\303\\251.dart")."""
    if len(text) < 2 or text[0] != '"' or text[-1] != '"':
        return text
    escapes = {'a': 7, 'b': 8, 't': 9, 'n': 10, 'v': 11, 'f': 12, 'r': 13, '"': 34, '\\': 92}
    body, out, i = text[1:-1], bytearray(), 0
    while i < len(body):
        char = body[i]
        if char == '\\' and i + 1 < len(body):
            octal = body[i + 1:i + 4]
            if re.fullmatch('[0-7]{3}', octal):
                out.append(int(octal, 8))
                i += 4
                continue
            out.append(escapes.get(body[i + 1], ord(body[i + 1])))
            i += 2
            continue
        out.extend(char.encode('utf-8'))
        i += 1
    return out.decode('utf-8', 'replace')


def parse_changes(text, prefix):
    changes = []
    for line in text.splitlines():
        if line.startswith('@@'):
            changes.append([])
        elif line.strip() and changes:
            path = unquote_git_path(line)
            if path.startswith(prefix):
                changes[-1].append(path[len(prefix):])
    return changes


LOCALITY_KEYS = ('changesScanned', 'changes', 'droppedOversized', 'changesWithoutFeatures',
                 'meanFeaturesPerChange', 'changesTouching3PlusFeatures', 'topCoChangedPairs')


def read_changes_file(path):
    try:
        with open(path, encoding='utf-8', errors='replace') as handle:
            return handle.read()
    except OSError as error:
        die(f'cannot read --changes-file {path}: {error.strerror or error}')


def change_locality(args, root, files, modmap, changes_text):
    """Feature spread of recent changes; a history that cannot be read leaves null fields and a reason."""
    if changes_text is not None:
        text, prefix = changes_text, ''
    else:
        reason = git_unavailable(root, args.no_git)
        if reason:
            return dict(dict.fromkeys(LOCALITY_KEYS), reason=reason)
        try:
            prefix = git(root, 'rev-parse', '--show-prefix').strip()
            # --diff-merges needs git 2.31; an older git fails here and is reported like any other failure.
            text = git(root, '-c', 'core.quotePath=false', 'log', '--first-parent', '--diff-merges=first-parent',
                       '-n', str(args.changes), '--name-only', '--format=@@%H', args.branch, '--')
        except GitError as error:
            return dict(dict.fromkeys(LOCALITY_KEYS), reason=f'git log failed: {error}')
    changes = parse_changes(text, prefix)
    touched, oversized, without = [], 0, 0
    for paths in changes:
        source = {path for path in paths if files.get(path, (0, ''))[1] == 'src'}
        if len(source) > args.max_changeset:
            oversized += 1
            continue
        features = sorted({owner[0].name for owner in map(modmap.owner, source)
                           if owner and owner[0].role == 'feature'})
        if features:
            touched.append(features)
        else:
            without += 1
    if not touched:
        return dict(dict.fromkeys(LOCALITY_KEYS), reason=f'no qualifying change in the window ({len(changes)} scanned, '
                                                         f'{oversized} oversized, {without} without features)')
    pairs = Counter(f'{a}<->{b}' for features in touched for i, a in enumerate(features) for b in features[i + 1:])
    return {
        'changesScanned': len(changes),
        'changes': len(touched),
        'droppedOversized': oversized,
        'changesWithoutFeatures': without,
        'meanFeaturesPerChange': sum(map(len, touched)) / len(touched),
        'changesTouching3PlusFeatures': pct(sum(len(f) >= 3 for f in touched), len(touched)),
        'topCoChangedPairs': ranked(pairs),
        'reason': None,
    }


def size_profile(args, src_lines, features):
    peak = None
    inventory = os.path.join(args.dir, 'inventory.json')
    if os.path.isfile(inventory):
        try:
            with open(inventory, encoding='utf-8') as handle:
                peak = (json.load(handle).get('git') or {}).get('peakMonthlyContributors')
        except (json.JSONDecodeError, AttributeError) as error:
            warn(f'{inventory} is unreadable ({error}); peakMonthlyContributors treated as unknown')
    if (src_lines < COMPACT_MAX_SRC_LINES and features <= COMPACT_MAX_FEATURES
            and (peak is None or peak <= COMPACT_MAX_PEAK_CONTRIBUTORS)):
        measured = 'compact'
    elif (src_lines >= LARGE_MIN_SRC_LINES or features >= LARGE_MIN_FEATURES
          or (peak is not None and peak >= LARGE_MIN_PEAK_CONTRIBUTORS)):
        measured = 'large'
    else:
        measured = 'standard'
    return {
        'measured': measured, 'override': args.profile, 'effective': args.profile or measured,
        'inputs': {'srcLines': src_lines, 'features': features, 'peakMonthlyContributors': peak},
        'drivers': sorted(set(args.driver or ())),
    }


def run_metrics(args):
    root = os.path.abspath(args.root)
    check_output_dir(root, args.dir, 'metrics')
    changes_text = read_changes_file(args.changes_file) if args.changes_file else None
    files = load_files(args.dir)
    modmap = load_module_map(args.dir, files)
    extraction = read_edges_in(args.dir, files) if args.lang == 'none' else EXTRACTORS[args.lang](root, files)
    with open_output(os.path.join(args.dir, 'edges.tsv')) as handle:
        handle.writelines(f'{source}\t{target}\t{site.line}\n'
                          for (source, target), site in sorted(extraction.edges.items()))
    edges = cross_module_edges(extraction, modmap)
    with open_output(os.path.join(args.dir, 'module_edges.tsv')) as handle:
        handle.write('source_file\tline\tsource_module\tsource_role\ttarget_file\ttarget_module\ttarget_role\tstatus\n')
        handle.writelines('\t'.join(map(str, row)) + '\n' for row in module_edge_rows(extraction, modmap, edges))
    coupling, violations, lists = coupling_metrics(edges, modmap)
    unmapped = sorted(path for path in files if modmap.owner(path) is None)
    src_files = [lines for lines, kind in files.values() if kind == 'src']
    features = sum(m.role == 'feature' for m in modmap.modules.values())
    unresolved = sorted(extraction.unresolved)
    metrics = {
        'totals': {
            'files': len(files), 'srcFiles': len(src_files), 'srcLines': sum(src_files),
            'genLines': sum(lines for lines, kind in files.values() if kind == 'gen'),
            'modules': len(modmap.modules), 'features': features,
            'edgesExtracted': len(extraction.edges), 'crossModuleEdges': len(edges),
            'unresolvedInternal': len(unresolved),
            'unresolvedExamples': [f'{source}:{line} {spec}' for source, line, spec in unresolved[:20]],
            'typeOnlyEdges': sum(site.type_only for site in extraction.edges.values()),
            'parseErrors': sorted(extraction.parse_errors)[:20], 'parseErrorFiles': len(extraction.parse_errors),
            'unmappedFiles': len(unmapped), 'unmappedExamples': unmapped[:20],
        },
        'structuralCoupling': coupling,
        'violations': violations,
        **lists,
        'modules': module_table(files, modmap, edges),
        'sizes': size_metrics(files, modmap),
        'platformImportsByModule': platform_imports(extraction.platform, modmap),
        'externalImportsByModule': external_imports(extraction.externals, modmap),
        'changeLocality': change_locality(args, root, files, modmap, changes_text),
        'profile': size_profile(args, sum(src_files), features),
    }
    error = None
    if not extraction.edges and len(src_files) >= 2:
        if unresolved:  # internal-looking imports that resolved to nothing: the inputs are wrong
            error = (f'no import edges between the {len(src_files)} source files and {len(unresolved)} unresolved internal '
                     'imports: check --lang, the inventory sources and the path aliases (tsconfig paths, pubspec names, '
                     'Python roots)')
            metrics['error'] = error
        else:  # the files import only packages: an empty graph is a finding, not a failure
            metrics['warning'] = (f'no import edges between the {len(src_files)} source files and no unresolved internal '
                                  'import: they import only external packages')
            warn(metrics['warning'])
    write_json(os.path.join(args.dir, 'metrics.json'), metrics)
    if unmapped:
        warn(f'{len(unmapped)} files match no module_map.tsv row, e.g. {", ".join(unmapped[:5])}')
    print(json.dumps({key: metrics[key] for key in ('totals', 'structuralCoupling')}, indent=2, sort_keys=True))
    if error:
        die(error)


def self_digest(text):
    # Line endings are normalised so a copy extracted in text mode on Windows (CRLF) still matches.
    text = text.replace('\r\n', '\n').replace('\r', '\n')
    neutral = SELF_DIGEST_LINE.sub("SELF_DIGEST = '" + '0' * 64 + "'", text, count=1)
    return hashlib.sha256(neutral.encode('utf-8')).hexdigest()


def check_integrity():
    with open(__file__, encoding='utf-8', newline='') as handle:
        actual = self_digest(handle.read())
    if actual != SELF_DIGEST:
        print(f'graph_metrics.py was altered (expected {SELF_DIGEST}, got {actual}); extract it again', file=sys.stderr)
        sys.exit(3)
